// remplissageTableau
void remplissageTableau(int T[],int N){

    printf("Donner la %de valeur:\n",(i+1));

 }
 // affichageTableau
 void affichageTableau(int T[],int N){
  int i;
  for(i=0; i<N; i++){
    printf("%d\t",T[i]);
  }
 }
 // triTableauCroissant
 void triTableauCroissant(int T[],int N){
 int i,j,tmp;
 for(i=0; i<N-1; i++){
        for(j=i+1; j<N; j++){
            if(T[i] > T[j]){
                tmp = T[i];
                T[i] = T[j];
                T[j] = tmp;
    }
   }
  }
 }

 // decalage a gauche
 void decalageGauche(int T[],int N,int K){
 int i,j,svg;
 for(i=0; i<K; i++){
// principe de decalge des elements du vecteur d'un rang a gauche
    svg = T[0];
    for(j=0; j<N-1;j++){
        T[j] = T[j+1];
    }
      T[N-1] = svg;
 }
 }
 // decalage a droite
void decalageDroite(int T[],int N,int K){
int i,j,svg;
 for(i=0; i<K; i++){
// principe de decalge des elements du vecteur d'un rang a droite
    svg = T[N-1];
    for(j=N-1; j>0; j--){
        T[j] = T[j-1];
    }
      T[0] = svg;
 }
}
void sommeTableau(int T[],int N){
 int i,som =0;
 for(i=0; i<N; i++){
    som += T[i];
 }
}
void produitTableau(int T[],int N){
 int i,prod =1;
 for(i=0; i<N; i++){
    prod *= T[i];
 }
}
void inverseTableau(int T[],int N){
int i,j,svg;
i=0;
j=N-1;
while(i<j){
    svg = T[i];
    T[i] = T[j];
    T[j] = svg;
    i++;
    j--;
}
}
